Setup — Planilha de Renovações

Guia completo para configurar o Google Sheets + Apps Script + alertas por email

1

Abrir o Editor de Scripts

Na planilha do Google Sheets:

  • Menu superior: Extensões → Apps Script
  • Vai abrir o editor de código em uma nova aba
  • Selecione todo o conteúdo do arquivo Código.gs (Ctrl+A / Cmd+A)
  • Apague tudo e cole todo o código do Passo 2 abaixo
2

Colar o Código Completo

Cole este código inteiro no editor do Apps Script (substituindo tudo):

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = getOrCreateSheet();
    var plan = PLANS[data.plano];

    if (!plan) {
      return jsonResponse({ status: 'error', message: 'Plano inválido' });
    }

    var now = new Date();
    var expDate = new Date(now);
    expDate.setDate(expDate.getDate() + plan.days);

    var daysLeft = calcDaysLeft(expDate);
    var status = getStatus(daysLeft);

    var phoneFormatted = formatPhone(data.whatsapp);

    // Usa o valor enviado pelo formulario (prioridade) ou o da tabela PLANS
    var finalPrice = (data.valor && !isNaN(data.valor)) ? Number(data.valor) : plan.price;
    var finalName = data.plano_nome || plan.name;

    sheet.appendRow([
      data.nome,
      data.email,
      phoneFormatted,
      finalName,
      formatCurrency(finalPrice),
      formatDate(now),
      formatDate(expDate),
      daysLeft,
      status,
      '',  // Observações
      ''   // Situação (vazio = ativo normal)
    ]);

    // Garantir que a célula do WhatsApp é texto
    var lastRow = sheet.getLastRow();
    sheet.getRange(lastRow, 3).setNumberFormat('@').setValue(phoneFormatted);

    // Colorir a linha recém-adicionada
    colorRow(sheet, lastRow, status, '');

    return jsonResponse({ status: 'ok' });
  } catch (error) {
    return jsonResponse({ status: 'error', message: error.toString() });
  }
}

function doGet(e) {
  var action = e && e.parameter ? e.parameter.action : '';
  var key = e && e.parameter ? e.parameter.key : '';

  if (action === 'teste' && key === ALERT_KEY) {
    sendTestEmail();
    return ContentService.createTextOutput('E-mail de teste enviado para: ' + ALERT_EMAILS.join(', '));
  }
  if (action === 'alerta' && key === ALERT_KEY) {
    sendDailyAlert();
    return ContentService.createTextOutput('Alerta diário executado agora.');
  }
  return ContentService.createTextOutput(
    'Smart Play — Sistema de Renovações ativo. v1.1'
  );
}

function setupSheet() {
  var sheet = getOrCreateSheet();

  var headers = [
    'Nome', 'Email', 'WhatsApp', 'Plano', 'Valor',
    'Data Ativação', 'Data Vencimento', 'Dias Restantes',
    'Status', 'Observações', 'Situação'
  ];

  var headerRange = sheet.getRange(1, 1, 1, headers.length);
  headerRange.setValues([headers]);
  headerRange.setBackground('#0F0808');
  headerRange.setFontColor('#FF6B6B');
  headerRange.setFontWeight('bold');
  headerRange.setFontFamily('Montserrat');
  headerRange.setHorizontalAlignment('center');

  // Largura das colunas
  sheet.setColumnWidth(1, 200);   // Nome
  sheet.setColumnWidth(2, 260);   // Email
  sheet.setColumnWidth(3, 170);   // WhatsApp
  sheet.setColumnWidth(4, 200);   // Plano
  sheet.setColumnWidth(5, 120);   // Valor
  sheet.setColumnWidth(6, 140);   // Data Ativação
  sheet.setColumnWidth(7, 140);   // Data Vencimento
  sheet.setColumnWidth(8, 130);   // Dias Restantes
  sheet.setColumnWidth(9, 120);   // Status
  sheet.setColumnWidth(10, 220);  // Observações
  sheet.setColumnWidth(11, 130);  // Situação

  // Congelar cabeçalho
  sheet.setFrozenRows(1);

  // Coluna WhatsApp como texto puro
  sheet.getRange('C:C').setNumberFormat('@');

  // Alinhar colunas ao centro
  sheet.getRange('E:I').setHorizontalAlignment('center');
  sheet.getRange('K:K').setHorizontalAlignment('center');

  // Dropdown de validação na coluna Situação
  var situacaoRange = sheet.getRange('K2:K1000');
  var regra = SpreadsheetApp.newDataValidation()
    .requireValueInList(['', 'Renovou', 'Cancelou'], true)
    .setAllowInvalid(false)
    .build();
  situacaoRange.setDataValidation(regra);

  Logger.log('Planilha configurada com sucesso! (Smart Play v1.0)');
}


// ══════════════════════════════════════════════════════════
//  3. ATUALIZAR STATUS DE TODOS OS CLIENTES
// ══════════════════════════════════════════════════════════

function updateAllStatus() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sheet) return;

  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;

  var numRows = lastRow - 1;
  var data = sheet.getRange(2, 1, numRows, TOTAL_COLUNAS).getValues();

  for (var i = 0; i < data.length; i++) {
    var situacao = String(data[i][COL.SITUACAO]).trim();

    // Linhas com Situação marcada: aplica cor especial e pula cálculo
    if (situacao === 'Renovou' || situacao === 'Cancelou') {
      colorRow(sheet, i + 2, '', situacao);
      continue;
    }

    var expValue = data[i][COL.DATA_VENCIMENTO];
    if (!expValue) continue;

    var expDate = parseDate(expValue);
    if (!expDate) continue;

    var daysLeft = calcDaysLeft(expDate);
    var status = getStatus(daysLeft);
    var row = i + 2;

    sheet.getRange(row, COL.DIAS_RESTANTES + 1).setValue(daysLeft);
    sheet.getRange(row, COL.STATUS + 1).setValue(status);

    colorRow(sheet, row, status, situacao);
  }

  Logger.log('Status atualizado para ' + numRows + ' clientes.');
}

function sendDailyAlert() {
  // Primeiro atualiza todos os status
  updateAllStatus();

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sheet) return;

  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;

  var data = sheet.getRange(2, 1, lastRow - 1, TOTAL_COLUNAS).getValues();

  var urgentes = [];    // 1-3 dias
  var renovar = [];     // 4-7 dias
  var vencidos = [];    // <= 0 dias (sem Situação marcada)
  var recuperar = [];   // Cancelou — tentar reativar

  var totalAtivos = 0;

  for (var i = 0; i < data.length; i++) {
    var situacao = String(data[i][COL.SITUACAO]).trim();

    // Renovou = ignorar completamente
    if (situacao === 'Renovou') continue;

    var cliente = {
      nome:       data[i][COL.NOME],
      email:      data[i][COL.EMAIL],
      whatsapp:   data[i][COL.WHATSAPP],
      plano:      data[i][COL.PLANO],
      valor:      data[i][COL.VALOR],
      ativacao:   data[i][COL.DATA_ATIVACAO],
      vencimento: data[i][COL.DATA_VENCIMENTO],
      dias:       data[i][COL.DIAS_RESTANTES],
      status:     data[i][COL.STATUS],
      obs:        data[i][COL.OBSERVACOES] || ''
    };

    // Cancelou = vai pra seção Recuperar
    if (situacao === 'Cancelou') {
      recuperar.push(cliente);
      continue;
    }

    // Cliente normal — classificar por status
    var status = data[i][COL.STATUS];

    if (status === 'Urgente')  urgentes.push(cliente);
    if (status === 'Renovar')  renovar.push(cliente);
    if (status === 'Vencido')  vencidos.push(cliente);
    if (status === 'Ativo')    totalAtivos++;
  }

  // Se não tem ninguém pra alertar, não envia
  if (urgentes.length === 0 && renovar.length === 0 && vencidos.length === 0 && recuperar.length === 0) {
    Logger.log('Nenhum cliente para alertar hoje.');
    return;
  }

  // Montar email HTML
  var html = buildEmailHTML(urgentes, renovar, vencidos, recuperar, totalAtivos, lastRow - 1);

  // Montar assunto
  var subject = 'Smart Play';

  if (urgentes.length > 0) {
    subject += ' | ' + urgentes.length + ' URGENTE(S)';
  }
  if (renovar.length > 0) {
    subject += ' | ' + renovar.length + ' para renovar';
  }
  if (vencidos.length > 0) {
    subject += ' | ' + vencidos.length + ' vencido(s)';
  }
  if (recuperar.length > 0) {
    subject += ' | ' + recuperar.length + ' para recuperar';
  }

  // Enviar para todos os emails
  for (var j = 0; j < ALERT_EMAILS.length; j++) {
    try {
      MailApp.sendEmail({
        to: ALERT_EMAILS[j],
        subject: subject,
        htmlBody: html
      });
    } catch (err) {
      Logger.log('Erro ao enviar para ' + ALERT_EMAILS[j] + ': ' + err);
    }
  }

  Logger.log('Alerta enviado para ' + ALERT_EMAILS.length + ' emails.');
}

function createTriggers() {
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    ScriptApp.deleteTrigger(triggers[i]);
  }

  ScriptApp.newTrigger('sendDailyAlert')
    .timeBased()
    .atHour(8)
    .everyDays(1)
    .inTimezone('America/Sao_Paulo')
    .create();

  Logger.log('Trigger criado: alerta diário às 8h (Brasília).');
}


// ══════════════════════════════════════════════════════════

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    setupSheet();
  }
  return sheet;
}

function calcDaysLeft(expDate) {
  var today = new Date();
  today.setHours(0, 0, 0, 0);
  var exp = new Date(expDate);
  exp.setHours(0, 0, 0, 0);
  return Math.ceil((exp - today) / (1000 * 60 * 60 * 24));
}

function getStatus(daysLeft) {
  if (daysLeft <= 0) return 'Vencido';
  if (daysLeft <= 3) return 'Urgente';
  if (daysLeft <= 7) return 'Renovar';
  return 'Ativo';
}

function colorRow(sheet, row, status, situacao) {
  var range = sheet.getRange(row, 1, 1, TOTAL_COLUNAS);

  // Se tem Situação marcada, usa cor especial
  if (situacao === 'Renovou') {
    range.setBackground('#E3F2FD');  // Azul claro
    range.setFontColor('#1565C0');
    return;
  }
  if (situacao === 'Cancelou') {
    range.setBackground('#FCE4EC');  // Rosa claro
    range.setFontColor('#AD1457');
    return;
  }

  // Cores padrão por status
  var colors = {
    'Ativo':   '#E8F5E9',
    'Renovar': '#FFF9C4',
    'Urgente': '#FFCDD2',
    'Vencido': '#E0E0E0'
  };
  var fontColors = {
    'Ativo':   '#1B5E20',
    'Renovar': '#F57F17',
    'Urgente': '#C62828',
    'Vencido': '#616161'
  };
  range.setBackground(colors[status] || '#FFFFFF');
  range.setFontColor(fontColors[status] || '#000000');
}

function formatDate(date) {
  return Utilities.formatDate(new Date(date), 'America/Sao_Paulo', 'dd/MM/yyyy');
}

function formatCurrency(value) {
  return 'R$ ' + value.toFixed(2).replace('.', ',');
}

function formatPhone(phone) {
  var p = String(phone).replace(/\D/g, '');
  if (p.length === 13 && p.substring(0,2) === '55') {
    p = p.substring(2);
  }
  if (p.length === 11) {
    return '(' + p.substring(0,2) + ') ' + p.substring(2,7) + '-' + p.substring(7);
  }
  if (p.length === 10) {
    return '(' + p.substring(0,2) + ') ' + p.substring(2,6) + '-' + p.substring(6);
  }
  return p;
}

function parseDate(value) {
  if (value instanceof Date) return value;
  if (typeof value === 'string') {
    var parts = value.split('/');
    if (parts.length === 3) {
      return new Date(parts[2], parts[1] - 1, parts[0]);
    }
  }
  return null;
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function clienteCard(c) {
  var whatsNum = String(c.whatsapp).replace(/\D/g, '');
  if (whatsNum.length <= 11) {
    whatsNum = '55' + whatsNum;
  }
  var whatsLink = 'https://wa.me/' + whatsNum;
  var html = '';
  html += '<div style="background:#fff;padding:12px 16px;border-radius:8px;margin:8px 0;font-size:13px;">';
  html += '<strong style="font-size:14px;">' + c.nome + '</strong>';
  html += '<br>Plano: ' + c.plano + ' | ' + c.valor;
  html += '<br>WhatsApp: <a href="' + whatsLink + '">' + c.whatsapp + '</a>';
  html += '<br>Vencimento: ' + c.vencimento + ' (' + c.dias + ' dias)';
  if (c.obs) {
    html += '<br><em style="color:#888;">Obs: ' + c.obs + '</em>';
  }
  html += '</div>';
  return html;
}

Salvar e Rodar a Configuração

Depois de colar o código:

  • Clique em Ctrl+S (ou Cmd+S no Mac) para salvar
  • Selecione a função setupSheet no dropdown
  • Clique em Executar (play)
  • Se pedir autorização: Revisar permissões → sua conta → Avançado → Ir para Smart Play (inseguro) → Permitir
Por que aparece "inseguro"? É normal para scripts próprios. O Google avisa porque não foi verificado por eles — mas é o SEU script, na SUA planilha.

Depois, selecione createTriggers e execute. Isso agenda o alerta diário automático às 8h.

Formulário HTMLColeta nome, email, WhatsApp (DDD + número) e plano
Meta PixelDispara evento Purchase com valor e plano
Webhook → SheetsEnvia dados para a planilha automaticamente
Cálculo automáticoData de vencimento e dias restantes
Cores por statusVerde (ativo), Amarelo (renovar), Vermelho (urgente), Cinza (vencido)
Coluna SituaçãoDropdown: Renovou (azul) ou Cancelou (rosa)
Email diário (8h)Urgentes + Renovar + Vencidos + Recuperar (cancelados)
Link WhatsAppCada cliente no email tem link direto wa.me
🎨

Guia de Cores da Planilha

CorStatusSituaçãoAparece no email?
■ Verde Ativo (>7 dias) — Não (está tranquilo)
■ Amarelo Renovar (4-7 dias) — Sim — seção Renovar
■ Vermelho Urgente (1-3 dias) — Sim — seção Urgente
■ Cinza Vencido — Sim — seção Vencidos
■ Azul — Renovou Não (resolvido)
■ Rosa — Cancelou Sim — seção Recuperar
★

Fluxo Operacional Diário

  • Venda no WhatsApp: Após ativar o cliente, envia o link da página "Ativar Cadastro"
  • Cliente preenche: Dados vão para a planilha + Pixel dispara Purchase
  • 8h da manhã: Email automático com Urgentes, Renovar, Vencidos e Recuperar
  • Equipe age: Prioridade: vermelho → amarelo → cinza → rosa
  • Cliente renovou: Marca "Renovou" na linha antiga (fica azul, sai do email)
  • Cliente cancelou: Marca "Cancelou" na linha (fica rosa, vai pra seção Recuperar)
  • Recuperação: Ao longo da semana, tenta reativar os rosa com promoção ou contato
